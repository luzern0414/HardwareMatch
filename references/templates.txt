# 01-system/02-research.md：调研记录格式与最低门槛
=== template: research.md ===
# Research

Every completed record below must come from an actual LuzCrawl run for this session. Keep the raw local snapshot and its SHA-256. A copied URL, a hand-written status, or an unverifiable summary cannot support an architecture decision.

<!--
### REF-001
- Source name: 中文来源名
- URL: https://example.com/page
- Registrable domain: example.com
- Access date: YYYY-MM-DD
- HTTP status: 200
- Page title: 页面标题
- LuzCrawl task ID: LC-YYYYMMDD-001
- LuzCrawl status: 成功
- Evidence category: 页面结构
- Visible evidence: 页面中实际可见且与本会话有关的事实
- Supported functions: F-001，或待确认功能描述
- Business flow or module: 已批准模块名
- Borrowable: 可借鉴的结构或规则
- Not copyable: 品牌、文案、图像、字体、布局等禁止照搬项
- Snapshot path: 01-system/crawl-snapshots/LC-YYYYMMDD-001.md
- Snapshot sha256: 64位小写十六进制摘要
-->

Minimum gate: at least 12 unique URLs across 5 registrable domains, with at least 3 page-structure, 3 interaction-flow, and 2 component-state records. Research only frontend references: page layouts, navigation, interaction flows, component states, and visual treatment. Business rules, regulations, payment callbacks, and other backend behavior belong to Luz Dev and are not research sources here. Each snapshot must be a session-local file whose actual SHA-256 matches the record.
=== end ===

// List pagination (admin-design.html 分页): default 20, 10/20/50/100 条/页, page and page_size kept in the address,
// the chosen page size remembered on this device, back to page 1 after a size change, more than 7 pages fold into an ellipsis.
(function () {
  var SIZES = [10, 20, 50, 100], total = 86, states = ["待处理", "进行中", "已完成"], owners = ["张敏", "李强", "王芳", "赵磊"], data = [];
  for (var i = 1; i <= total; i++) {
    data.push(["示例记录 " + String(i).padStart(3, "0"), states[i % 3], owners[i % 4]]);
  }
  var rows = document.getElementById("list-rows"), wrap = document.getElementById("list-wrap");
  var sizeSelect = document.getElementById("pg-size"), pages = document.getElementById("pg-pages");
  var prev = document.getElementById("pg-prev"), next = document.getElementById("pg-next");
  var jump = document.getElementById("pg-jump"), jumpInput = document.getElementById("pg-jump-input");
  var params = new URLSearchParams(location.search), stored = null;
  try { stored = localStorage.getItem("list_page_size"); } catch (e) {}
  var size = Number(params.get("page_size") || stored);
  if (SIZES.indexOf(size) < 0) { size = 20; }
  var page = Math.max(1, parseInt(params.get("page"), 10) || 1);
  function count() { return Math.max(1, Math.ceil(total / size)); }
  function numbers(last) {
    if (last <= 7) { return Array.from({ length: last }, function (_, k) { return k + 1; }); }
    var start = Math.max(2, page - 2), end = Math.min(last - 1, page + 2), list = [1];
    if (page <= 4) { start = 2; end = 5; }
    if (page >= last - 3) { start = last - 4; end = last - 1; }
    if (start > 2) { list.push("…"); }
    for (var n = start; n <= end; n++) { list.push(n); }
    if (end < last - 1) { list.push("…"); }
    list.push(last);
    return list;
  }
  function pageButton(n) {
    var b = document.createElement("button");
    b.type = "button";
    b.textContent = n;
    b.dataset.actionId = prev.dataset.actionId;
    b.dataset.pageId = prev.dataset.pageId;
    if (n === page) { b.setAttribute("aria-current", "page"); }
    b.addEventListener("click", function () { go(n); });
    return b;
  }
  function render() {
    var last = count();
    page = Math.min(page, last);
    rows.innerHTML = "";
    data.slice((page - 1) * size, page * size).forEach(function (row) {
      var tr = document.createElement("tr");
      row.forEach(function (value) { var td = document.createElement("td"); td.textContent = value; tr.appendChild(td); });
      rows.appendChild(tr);
    });
    document.getElementById("pg-total").textContent = total;
    pages.innerHTML = "";
    numbers(last).forEach(function (n) {
      if (n === "…") { var gap = document.createElement("span"); gap.className = "pg-gap"; gap.textContent = "…"; pages.appendChild(gap); }
      else { pages.appendChild(pageButton(n)); }
    });
    // One page keeps 共 N 条 and hides the page controls; more than five pages add 跳至.
    prev.hidden = next.hidden = pages.hidden = last === 1;
    jump.hidden = last <= 5;
    prev.disabled = page === 1;
    next.disabled = page === last;
    sizeSelect.value = String(size);
    params.set("page", page);
    params.set("page_size", size);
    try { history.replaceState(null, "", "?" + params.toString() + location.hash); } catch (e) {}
  }
  function go(n) { page = Math.min(Math.max(1, n), count()); render(); wrap.scrollTop = 0; }
  sizeSelect.addEventListener("change", function () {
    size = Number(sizeSelect.value);
    try { localStorage.setItem("list_page_size", size); } catch (e) {}
    page = 1;
    render();
    wrap.scrollTop = 0;
  });
  prev.addEventListener("click", function () { go(page - 1); });
  next.addEventListener("click", function () { go(page + 1); });
  jumpInput.addEventListener("change", function () { var n = parseInt(jumpInput.value, 10); if (n) { go(n); } jumpInput.value = ""; });
  render();
})();
</script>

(function () {
  function label(s) {
    if (s < 60) { return s + " 秒"; }
    if (s < 3600) { return Math.floor(s / 60) + " 分 " + String(s % 60).padStart(2, "0") + " 秒"; }

    ticking.forEach(function (node) {

      node.setAttribute("data-elapsed", s);
      node.textContent = (node.textContent.indexOf("已用") === 0 ? "已用 " : "") + label(s);
